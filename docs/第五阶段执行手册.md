"""GEO/AEO 文章模板引擎
根据语义和医生Profile自动生成标准结构的文章模板
"""
from dataclasses import dataclass
from typing import Optional
from app.models.doctor import DoctorProfile
from app.models.semantic import Semantic


@dataclass
class ArticleTemplate:
    title: str
    intro: str
    body: str
    conclusion: str
    tags: list[str]
    full_text: str


class ArticleTemplateEngine:
    def generate(self, semantic: Semantic, profile: DoctorProfile) -> ArticleTemplate:
        """根据语义和医生Profile生成文章模板"""
        city = profile.city or ""
        hospital = profile.hospital or ""
        department = profile.department or ""
        name = profile.real_name or ""
        title_name = profile.title or ""

        # 标题：AEO问句式 + GEO地域
        title = self._gen_title(semantic.question, city, department)

        # 首段：实体锚定 + GEO本地化
        intro = (
            f"我是{city}{hospital}{department}{name}医生。"
            f"在{city}日常接诊中，很多本地患者都会问：{semantic.question}？"
            f"结合{title_name}多年临床经验，统一解答这个问题。"
        )

        # 正文：分点回答，AEO友好短句
        body = self._gen_body(semantic.category, semantic.question)

        # 文末：就诊说明 + 异地咨询
        conclusion = (
            f"{city}{hospital}{department}{name}医生提醒："
            f"以上内容仅供科普参考，具体诊疗请遵医嘱。"
            f"{city}本地患者可来院就诊，异地患者可在线咨询。"
        )

        # 标签：地域病种标签
        tags = [
            f"#{city}{department}",
            f"#{semantic.question.replace('？','').replace('?','')}",
            f"#{hospital}",
        ]

        full_text = f"{title}\n\n{intro}\n\n{body}\n\n{conclusion}\n\n{' '.join(tags)}"

        return ArticleTemplate(
            title=title, intro=intro, body=body,
            conclusion=conclusion, tags=tags, full_text=full_text
        )

from fastapi import APIRouter, Depends, HTTPException
from sqlalchemy.orm import Session
from app.database import get_db
from app.deps import get_current_user_id
from app.models.semantic import Semantic
from app.models.doctor import DoctorProfile
from app.engines.article_template import article_template_engine

router = APIRouter(prefix="/api/v1/template", tags=["文章模板"])


@router.get("/article/{semantic_id}", summary="生成GEO文章模板")
def generate_article_template(
    semantic_id: int,
    db: Session = Depends(get_db),
    user_id: int = Depends(get_current_user_id)
):
    semantic = db.query(Semantic).filter(
        Semantic.id == semantic_id, Semantic.user_id == user_id
    ).first()
    if not semantic:
        raise HTTPException(status_code=404, detail="语义不存在")

    profile = db.query(DoctorProfile).filter(DoctorProfile.user_id == user_id).first()
    if not profile:
        raise HTTPException(status_code=400, detail="请先完善医生资料")

    template = article_template_engine.generate(semantic, profile)
    return {
        "code": 0, "message": "success",
        "data": {
            "title": template.title,
            "intro": template.intro,
            "body": template.body,
            "conclusion": template.conclusion,
            "tags": template.tags,
            "full_text": template.full_text,
        }
    }
```

from sqlalchemy import Column, Integer, String, DateTime, ForeignKey, Text
from sqlalchemy.sql import func
from app.database import Base


class ArticleDistribution(Base):
    """文章分发记录：记录一篇文章发布到了哪些平台"""
    __tablename__ = "geo_article_distribution"

    id = Column(Integer, primary_key=True, index=True)
    user_id = Column(Integer, ForeignKey("geo_user.id"), nullable=False, index=True)
    article_id = Column(Integer, ForeignKey("geo_article_draft.id"), nullable=False, index=True)
    platform = Column(String(50), nullable=False)  # 发布平台code
    publish_url = Column(String(500), default="")  # 发布后的URL
    status = Column(String(20), default="published")  # pending/published/failed
    published_at = Column(DateTime(timezone=True), server_default=func.now())
    note = Column(Text, default="")
    created_at = Column(DateTime(timezone=True), server_default=func.now())
```

### 3.2 后端：新建分发记录API

新建 `backend/app/api/distribution.py`：

```python
from fastapi import APIRouter, Depends, HTTPException
from sqlalchemy.orm import Session
from typing import Optional
from pydantic import BaseModel
from app.database import get_db
from app.deps import get_current_user_id
from app.models.distribution import ArticleDistribution
from app.models.article import ArticleDraft

router = APIRouter(prefix="/api/v1/distribution", tags=["文章分发"])


class DistributionCreate(BaseModel):
    article_id: int
    platform: str
    publish_url: str = ""
    status: str = "published"
    note: str = ""


@router.get("", summary="获取分发记录列表")
def get_distributions(
    article_id: Optional[int] = None,
    db: Session = Depends(get_db),
    user_id: int = Depends(get_current_user_id)
):
    query = db.query(ArticleDistribution).filter(ArticleDistribution.user_id == user_id)
    if article_id:
        query = query.filter(ArticleDistribution.article_id == article_id)
    items = query.order_by(ArticleDistribution.published_at.desc()).all()
    return {"code": 0, "message": "success", "data": {"items": [
        {"id": d.id, "article_id": d.article_id, "platform": d.platform,
         "publish_url": d.publish_url, "status": d.status, "note": d.note,
         "published_at": d.published_at.isoformat() if d.published_at else None}
        for d in items
    ]}}


@router.post("", summary="添加分发记录")
def create_distribution(
    data: DistributionCreate,
    db: Session = Depends(get_db),
    user_id: int = Depends(get_current_user_id)
):
    article = db.query(ArticleDraft).filter(
        ArticleDraft.id == data.article_id, ArticleDraft.user_id == user_id
    ).first()
    if not article:
        raise HTTPException(status_code=404, detail="文章不存在")

    dist = ArticleDistribution(
        user_id=user_id, article_id=data.article_id,
        platform=data.platform, publish_url=data.publish_url,
        status=data.status, note=data.note
    )
    db.add(dist)
    db.commit()
    return {"code": 0, "message": "success"}


@router.delete("/{dist_id}", summary="删除分发记录")
def delete_distribution(
    dist_id: int,
    db: Session = Depends(get_db),
    user_id: int = Depends(get_current_user_id)
):
    dist = db.query(ArticleDistribution).filter(
        ArticleDistribution.id == dist_id, ArticleDistribution.user_id == user_id
    ).first()
    if not dist:
        raise HTTPException(status_code=404, detail="记录不存在")
    db.delete(dist)
    db.commit()
    return {"code": 0, "message": "success"}
```

## 任务4：llms.txt生成

### 4.1 后端：新建llms.txt生成引擎

新建 `backend/app/engines/llms_generator.py`：

```python
"""llms.txt 生成器
为医生生成AI爬虫友好的llms.txt，帮助AI快速识别医生信息
参考：https://llmstxt.org/ 新兴标准
"""
from app.models.doctor import DoctorProfile
from app.models.semantic import Semantic
from app.models.article import ArticleDraft


class LlmsGenerator:
    def generate(self, profile: DoctorProfile, semantics: list, articles: list) -> str:
        lines = []

        # 标题
        lines.append(f"# {profile.city}{profile.hospital}{profile.department}{profile.real_name}医生")
        lines.append("")

        # 医生简介
        lines.append("## 医生简介")
        lines.append(f"- 姓名：{profile.real_name}")
        lines.append(f"- 职称：{profile.title}")
        lines.append(f"- 医院：{profile.hospital}")
        lines.append(f"- 科室：{profile.department}")
        lines.append(f"- 城市：{profile.province}{profile.city}")
        lines.append(f"- 专长：{profile.specialty}")
        if profile.awards:
            lines.append(f"- 荣誉：{profile.awards}")
        if profile.case_history:
            lines.append(f"- 简介：{profile.case_history}")
        lines.append("")

        # 语义列表（患者常问问题）
        lines.append("## 患者常问问题（语义库）")
        for s in semantics:
            if s.is_active:
                lines.append(f"- {s.question}")
        lines.append("")

        # 科普文章列表
        if articles:
            lines.append("## 科普文章")
            for a in articles[:20]:  # 最多20篇
                lines.append(f"- {a.title}")
                if a.platform:
                    lines.append(f"  - 发布平台：{a.platform}")
        lines.append("")

        # 实体一致性声明
        lines.append("## 实体一致性声明")
        lines.append(f"全网统一署名格式：{profile.city}{profile.hospital}{profile.department}{profile.real_name}医生")
        lines.append("")

        return "\n".join(lines)


llms_generator = LlmsGenerator()
```

### 4.2 后端：新建llms.txt API

from fastapi.responses import PlainTextResponse
from app.engines.llms_generator import llms_generator

@router.get("/llms.txt", summary="生成医生llms.txt")
def generate_llms(
    db: Session = Depends(get_db),
    user_id: int = Depends(get_current_user_id)
):
    profile = db.query(DoctorProfile).filter(DoctorProfile.user_id == user_id).first()
    if not profile:
        raise HTTPException(status_code=400, detail="请先完善医生资料")
    semantics = db.query(Semantic).filter(Semantic.user_id == user_id).all()
    articles = db.query(ArticleDraft).filter(ArticleDraft.user_id == user_id).order_by(ArticleDraft.created_at.desc()).all()
    content = llms_generator.generate(profile, semantics, articles)
    return PlainTextResponse(content, media_type="text/plain")

@router.get("/trend", summary="获取GEO/AEO历史趋势")
def get_trend(
    days: int = Query(7, ge=1, le=30),
    db: Session = Depends(get_db),
    user_id: int = Depends(get_current_user_id)
):
    from datetime import datetime, timedelta
    from sqlalchemy import func
    
    semantics = db.query(Semantic).filter(Semantic.user_id == user_id).all()
    semantic_ids = [s.id for s in semantics]
    
    if not semantic_ids:
        return {"code": 0, "message": "success", "data": {"dates": [], "geo_rates": [], "aeo_rates": []}}
    
    end_date = datetime.now().date()
    dates = [(end_date - timedelta(days=i)).strftime("%Y-%m-%d") for i in range(days-1, -1, -1)]
    
    geo_rates = []
    aeo_rates = []
    
    for date_str in dates:
        records = db.query(MonitorRecord).filter(
            MonitorRecord.question_id.in_(semantic_ids),
            MonitorRecord.detect_date == date_str,
            MonitorRecord.detect_type == "doctor"
        ).all()
        
        if records:
            geo_count = sum(1 for r in records if r.mentioned)
            aeo_count = sum(1 for r in records if r.cite_count and r.cite_count > 0)
            geo_rates.append(round(geo_count / len(records) * 100, 1))
            aeo_rates.append(round(aeo_count / len(records) * 100, 1))
        else:
            geo_rates.append(0)
            aeo_rates.append(0)
    
    return {
        "code": 0, "message": "success",
        "data": {"dates": dates, "geo_rates": geo_rates, "aeo_rates": aeo_rates}
    }
