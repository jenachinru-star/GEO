### 2.1 新建通用适配器

**新建文件：`backend/app/adapters/openai_compatible.py`**

```python
"""通用OpenAI兼容平台适配器
所有OpenAI兼容格式的平台（GLM、MiniMax、豆包、通义等）都继承这个类，
只需要指定platform_code，不需要重复写query和_extract_answer逻辑。
"""
from openai import OpenAI
from typing import Optional
from app.adapters.base import BaseAdapter, QueryResult
import traceback
import logging

logger = logging.getLogger(__name__)


class OpenAICompatibleAdapter(BaseAdapter):
    """OpenAI兼容格式通用适配器"""

    platform_code: str = "generic"  # 子类覆盖

    def __init__(self, api_key: str, base_url: str, model: str,
                 temperature: float = 0.3, max_tokens: int = 2000):
        super().__init__(api_key, base_url, model, temperature, max_tokens)
        self.client = OpenAI(
            api_key=api_key,
            base_url=base_url,
            timeout=120
        )

    def _extract_answer(self, response) -> str:
        """从响应中提取回答文本，处理各种情况"""
        try:
            if not response or not response.choices or len(response.choices) == 0:
                logger.warning(f"{self.platform_code}响应无choices: {response}")
                return ""

            message = response.choices[0].message

            # 优先取content
            content = message.content
            if content and content.strip():
                return content.strip()

            # 如果content为空，检查是否有reasoning_content（思考模型）
            reasoning = getattr(message, 'reasoning_content', None)
            if reasoning and reasoning.strip():
                logger.info(f"{self.platform_code}使用reasoning_content作为回答")
                return reasoning.strip()

            if hasattr(response, 'model_dump'):
                raw = response.model_dump()
                logger.warning(f"{self.platform_code}响应结构: {str(raw)[:500]}")

            logger.warning(f"{self.platform_code}回答为空，message: {message}")
            return ""

        except Exception as e:
            logger.error(f"提取{self.platform_code}回答失败: {e}")
            return ""

    def query(self, question: str, system_prompt: Optional[str] = None) -> QueryResult:
        try:
            logger.info(f"{self.platform_code}请求: model={self.model}, temp={self.temperature}, question={question[:50]}")

            messages = []
            if system_prompt:
                messages.append({"role": "system", "content": system_prompt})
            messages.append({"role": "user", "content": question})

            response = self.client.chat.completions.create(
                model=self.model,
                messages=messages,
                temperature=self.temperature,
                max_tokens=self.max_tokens
            )

            answer = self._extract_answer(response)

            if answer:
                logger.info(f"{self.platform_code}回答长度: {len(answer)}")
                return QueryResult(
                    platform_code=self.platform_code,
                    answer=answer,
                    success=True,
                    raw_response=response.model_dump() if hasattr(response, 'model_dump') else None
                )
            else:
                return QueryResult(
                    platform_code=self.platform_code,
                    answer="",
                    success=False,
                    error="模型返回空回答（可能是思考模型或内容审核）"
                )

        except Exception as e:
            error_msg = f"{type(e).__name__}: {str(e)}\n{traceback.format_exc()}"
            logger.error(f"{self.platform_code}调用异常: {error_msg[:500]}")
            return QueryResult(
                platform_code=self.platform_code,
                answer="",
                success=False,
                error=error_msg
            )

    def get_platform_code(self) -> str:
        return self.platform_code
```

```python
    @staticmethod
    def export_doctor_report(db: Session, user_id: int) -> bytes:
        """导出医生个人GEO报告（多Sheet）

        Sheet1: 医生Profile
        Sheet2: 语义库列表
        Sheet3: 文章列表
        Sheet4: GEO/AEO监测记录
        """
        from app.models.doctor import DoctorProfile
        from app.models.semantic import Semantic
        from app.models.article import ArticleDraft
        from app.models.monitor import MonitorRecord

        wb = Workbook()

        # Sheet1: 医生Profile
        ws1 = wb.active
        ws1.title = "医生资料"
        ws1.append(["字段", "值"])
        ExportService._style_header(ws1, 1, 2)
        profile = db.query(DoctorProfile).filter(DoctorProfile.user_id == user_id).first()
        if profile:
            fields = [
                ("姓名", profile.real_name),
                ("职称", profile.title),
                ("城市", profile.city),
                ("医院", profile.hospital),
                ("科室", profile.department),
                ("专长", profile.specialty),
                ("自我介绍", profile.self_intro_template),
            ]
            for k, v in fields:
                ws1.append([k, v])
        else:
            ws1.append(["未完善医生资料", ""])

        # Sheet2: 语义库
        ws2 = wb.create_sheet("语义库")
        ws2.append(["ID", "语义问题", "分类", "是否启用", "是否付费", "GEO率(%)", "AEO率(%)", "文章数"])
        ExportService._style_header(ws2, 1, 8)
        semantics = db.query(Semantic).filter(Semantic.user_id == user_id).all()
        for s in semantics:
            ws2.append([s.id, s.question, s.category, "是" if s.is_active else "否",
                       "是" if s.is_paid else "否", s.geo_rate, s.aeo_rate, s.article_count])

        # Sheet3: 文章列表
        ws3 = wb.create_sheet("我的文章")
        ws3.append(["ID", "标题", "关联语义ID", "发布平台", "状态", "实体一致", "创建时间"])
        ExportService._style_header(ws3, 1, 7)
        articles = db.query(ArticleDraft).filter(ArticleDraft.user_id == user_id).all()
        platform_names = {"wechat": "微信公众号", "xiaohongshu": "小红书", "zhihu": "知乎",
                         "video": "视频号", "douyin": "抖音", "other": "其他"}
        status_names = {"draft": "草稿", "published": "已发布"}
        for a in articles:
            ws3.append([a.id, a.title, a.semantic_id,
                       platform_names.get(a.platform, a.platform),
                       status_names.get(a.status, a.status),
                       "是" if a.entity_consistent else "否",
                       str(a.created_at) if a.created_at else ""])

        # Sheet4: GEO/AEO监测记录
        ws4 = wb.create_sheet("GEO-AEO监测")
        ws4.append(["ID", "语义ID", "平台", "检测日期", "GEO推荐次数", "AEO引用次数", "AEO置信度"])
        ExportService._style_header(ws4, 1, 7)
        records = db.query(MonitorRecord).filter(
            MonitorRecord.detect_type == "doctor",
            MonitorRecord.semantic_id.in_([s.id for s in semantics]) if semantics else False
        ).all()
        for r in records:
            ws4.append([r.id, r.semantic_id, r.platform_code, r.detect_date,
                       r.recommend_count or 0, r.cite_count or 0, r.aeo_confidence or 0])

        # 调整列宽
        for ws in [ws1, ws2, ws3, ws4]:
            for col in ws.columns:
                max_length = 0
                column = col[0].column_letter
                for cell in col:
                    try:
                        if len(str(cell.value)) > max_length:
                            max_length = len(str(cell.value))
                    except:
                        pass
                ws.column_dimensions[column].width = min(max_length + 2, 50)

        output = io.BytesIO()
        wb.save(output)
        return output.getvalue()
```

**新建文件：`backend/scripts/clean_test_data.py`**

```python
"""清理测试数据脚本
删除测试用户及其关联数据，用于上线前清理。
使用方法：python scripts/clean_test_data.py
"""
import sys
import os
sys.path.insert(0, os.path.dirname(os.path.dirname(os.path.abspath(__file__))))

from app.database import SessionLocal
from app.models.user import User
from app.models.doctor import DoctorProfile
from app.models.semantic import Semantic
from app.models.article import ArticleDraft
from app.models.monitor import MonitorRecord

TEST_USERNAMES = ["test_doctor", "test_doctor2", "test"]

def clean():
    db = SessionLocal()
    try:
        test_users = db.query(User).filter(User.username.in_(TEST_USERNAMES)).all()
        if not test_users:
            print("未找到测试用户，无需清理")
            return

        for user in test_users:
            print(f"清理用户: {user.username} (id={user.id})")
            # 删除关联数据
            db.query(MonitorRecord).filter(
                MonitorRecord.semantic_id.in_(
                    db.query(Semantic.id).filter(Semantic.user_id == user.id)
                )
            ).delete(synchronize_session=False)
            db.query(ArticleDraft).filter(ArticleDraft.user_id == user.id).delete()
            db.query(Semantic).filter(Semantic.user_id == user.id).delete()
            db.query(DoctorProfile).filter(DoctorProfile.user_id == user.id).delete()
            db.query(User).filter(User.id == user.id).delete()

        db.commit()
        print(f"已清理 {len(test_users)} 个测试用户及其关联数据")
    except Exception as e:
        db.rollback()
        print(f"清理失败: {e}")
        raise
    finally:
        db.close()

if __name__ == "__main__":
    confirm = input("确认清理测试数据？(yes/no): ")
    if confirm.lower() == "yes":
        clean()
    else:
        print("已取消")
```

## 8. 坑表（必须避开）

| # | 坑 | 后果 | 正确做法 |
|---|-----|------|----------|
| 1 | 重写glm.py/minimax.py时忘记保留类名 | factory.py import失败 | 类名必须还是GLMAdapter和MiniMaxAdapter，只改继承关系 |
| 2 | OpenAICompatibleAdapter的platform_code是类变量 | 子类不覆盖就会都是"generic" | 子类必须设置`platform_code = "glm"`和`platform_code = "minimax"` |
| 3 | 给auth API加鉴权 | 登录接口无法访问 | auth.py的所有端点都不加get_current_user_id |
| 4 | 给旧API加鉴权后忘记前端已带token | 前端调用失败 | 前端axios拦截器已经自动加token，不需要改前端调用 |
| 5 | export_doctor_report里semantics为空时in_([])报错 | SQL错误 | 用`if semantics else False`处理空列表 |
| 6 | 清理测试数据时删MonitorRecord的条件写错 | 误删brand类型记录 | 只删semantic_id属于测试用户的doctor类型记录 |
| 7 | 前端导出按钮放在overview为null时可点 | 导出空报告 | 加`disabled={!overview}` |
| 8 | README部署文档里的命令路径写错 | 用户照着做失败 | 所有命令基于项目根目录，cd路径要准确 |
