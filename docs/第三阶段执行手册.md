## 2. 任务1：后端 — 实体一致性校验引擎

**新建文件：`backend/app/engines/entity_validator.py`**

```python
"""实体一致性校验引擎
核心逻辑：AI会因为医生信息在不同平台不一致而降权。
校验医生Profile和文章中的实体信息是否统一。
"""
from typing import List, Dict, Optional
from dataclasses import dataclass, field
import re


@dataclass
class ValidationIssue:
    """校验问题"""
    level: str  # error / warning / info
    field: str  # 哪个字段有问题
    message: str  # 问题描述
    suggestion: str  # 修复建议


@dataclass
class ValidationResult:
    """校验结果"""
    score: int  # 0-100
    issues: List[ValidationIssue] = field(default_factory=list)

    @property
    def passed(self) -> bool:
        return not any(i.level == "error" for i in self.issues)

    def to_dict(self) -> Dict:
        return {
            "score": self.score,
            "passed": self.passed,
            "issues": [
                {"level": i.level, "field": i.field, "message": i.message, "suggestion": i.suggestion}
                for i in self.issues
            ]
        }


class EntityValidator:
    """实体一致性校验器"""

    # 标准自我介绍格式：XX市XX医院XX科XX医生
    STANDARD_PATTERN = r'^(.+市)?(.+医院)(.+科)(.+医生)$'

    def validate_profile(self, profile) -> ValidationResult:
        """校验医生Profile的实体一致性

        Args:
            profile: DoctorProfile对象
        Returns:
            ValidationResult
        """
        issues: List[ValidationIssue] = []
        score = 100

        real_name = (profile.real_name or "").strip()
        hospital = (profile.hospital or "").strip()
        department = (profile.department or "").strip()
        city = (profile.city or "").strip()
        title = (profile.title or "").strip()
        self_intro = (profile.self_intro_template or "").strip()

        # 1. 四要素完整性检查
        if not real_name:
            issues.append(ValidationIssue("error", "real_name", "医生姓名为空", "请填写真实姓名"))
            score -= 25
        if not hospital:
            issues.append(ValidationIssue("error", "hospital", "医院名称为空", "请填写所在医院全称"))
            score -= 25
        if not department:
            issues.append(ValidationIssue("error", "department", "科室为空", "请填写所在科室"))
            score -= 20
        if not city:
            issues.append(ValidationIssue("warning", "city", "城市为空", "建议填写所在城市，有助于AI本地化推荐"))
            score -= 10

        # 2. 自我介绍格式检查
        if self_intro:
            # 检查是否包含四要素
            if real_name and real_name not in self_intro:
                issues.append(ValidationIssue("warning", "self_intro_template",
                    f"自我介绍中未包含医生姓名「{real_name}」",
                    "建议在自我介绍中包含真实姓名，格式如「XX市XX医院XX科XX医生」"))
                score -= 10
            if hospital and hospital not in self_intro:
                issues.append(ValidationIssue("warning", "self_intro_template",
                    f"自我介绍中未包含医院名称「{hospital}」",
                    "建议在自我介绍中包含医院全称"))
                score -= 10
            if department and department not in self_intro:
                issues.append(ValidationIssue("warning", "self_intro_template",
                    f"自我介绍中未包含科室「{department}」",
                    "建议在自我介绍中包含科室名称"))
                score -= 5
        else:
            issues.append(ValidationIssue("info", "self_intro_template",
                "未设置自我介绍模板",
                "建议设置统一的自我介绍模板，在所有平台使用相同表述"))
            score -= 5

        # 3. 医院名称规范性检查（是否含"医院"二字）
        if hospital and "医院" not in hospital and "中心" not in hospital and "诊所" not in hospital:
            issues.append(ValidationIssue("warning", "hospital",
                f"医院名称「{hospital}」可能不完整",
                "建议使用医院全称，如「XX市人民医院」"))
            score -= 5

        score = max(0, min(100, score))
        return ValidationResult(score=score, issues=issues)

    def validate_article(self, article, profile) -> ValidationResult:
        """校验文章中的实体一致性

        Args:
            article: ArticleDraft对象
            profile: DoctorProfile对象
        Returns:
            ValidationResult
        """
        issues: List[ValidationIssue] = []
        score = 100

        real_name = (profile.real_name or "").strip()
        hospital = (profile.hospital or "").strip()
        department = (profile.department or "").strip()

        title = (article.title or "").strip()
        content = (article.content or "").strip()
        full_text = f"{title} {content}"

        # 1. 文章是否关联语义
        if not article.semantic_id:
            issues.append(ValidationIssue("warning", "semantic_id",
                "文章未关联语义问题",
                "建议关联语义问题，便于追踪该语义下的内容产出"))
            score -= 10

        # 2. 文章中是否包含医生姓名
        if real_name:
            if real_name not in full_text:
                issues.append(ValidationIssue("warning", "content",
                    f"文章中未提及医生姓名「{real_name}」",
                    "建议在文章署名或正文中包含真实姓名，增强AI实体识别"))
                score -= 15
            else:
                # 检查是否有不一致的署名（如"张医生"vs"张三"）
                # 简单规则：如果有"X医生"格式但X不是real_name的姓，提示
                alias_pattern = r'([\u4e00-\u9fa5]{1,2})医生'
                aliases = re.findall(alias_pattern, full_text)
                for alias in set(aliases):
                    if alias != real_name[:len(alias)] and alias + "医生" != real_name:
                        issues.append(ValidationIssue("info", "content",
                            f"文章中发现署名「{alias}医生」，与真实姓名「{real_name}」可能不一致",
                            f"建议统一使用「{real_name}」或「{real_name[-2:]}医生」作为署名"))
                        score -= 5
                        break

        # 3. 文章中是否包含医院名称
        if hospital and hospital not in full_text:
            issues.append(ValidationIssue("warning", "content",
                f"文章中未提及医院名称「{hospital}」",
                "建议在文章中包含医院全称，增强实体锚定"))
            score -= 10

        # 4. 文章中是否包含科室
        if department and department not in full_text:
            issues.append(ValidationIssue("info", "content",
                f"文章中未提及科室「{department}」",
                "建议在文章中提及科室名称，有助于AI精准关联"))
            score -= 5

        # 5. 文章内容长度检查
        if len(content) < 100:
            issues.append(ValidationIssue("warning", "content",
                f"文章内容较短（{len(content)}字）",
                "建议文章正文不少于300字，内容越丰富AI越容易引用"))
            score -= 10

        score = max(0, min(100, score))
        return ValidationResult(score=score, issues=issues)


# 全局单例
entity_validator = EntityValidator()
```

```python
from fastapi import APIRouter, Depends, HTTPException, Query
from sqlalchemy.orm import Session
from typing import Optional

from app.database import get_db
from app.deps import get_current_user_id
from app.models.article import ArticleDraft
from app.models.semantic import Semantic
from app.models.doctor import DoctorProfile
from app.schemas import ArticleCreate, ArticleUpdate, ArticleOut
from app.engines.entity_validator import entity_validator

router = APIRouter(prefix="/api/v1/article", tags=["文章管理"])


@router.get("", summary="获取当前用户的文章列表")
def get_articles(
    semantic_id: Optional[int] = None,
    status: Optional[str] = None,
    page: int = Query(1, ge=1),
    page_size: int = Query(20, ge=1, le=100),
    db: Session = Depends(get_db),
    user_id: int = Depends(get_current_user_id)
):
    query = db.query(ArticleDraft).filter(ArticleDraft.user_id == user_id)
    if semantic_id:
        query = query.filter(ArticleDraft.semantic_id == semantic_id)
    if status:
        query = query.filter(ArticleDraft.status == status)
    total = query.count()
    items = query.order_by(ArticleDraft.created_at.desc())\
        .offset((page - 1) * page_size).limit(page_size).all()
    return {
        "code": 0, "message": "success",
        "data": {
            "total": total,
            "items": [ArticleOut.model_validate(i).model_dump() for i in items],
            "page": page, "page_size": page_size
        }
    }


@router.post("", summary="创建文章记录")
def create_article(
    data: ArticleCreate,
    db: Session = Depends(get_db),
    user_id: int = Depends(get_current_user_id)
):
    # 验证语义属于当前用户
    semantic = db.query(Semantic).filter(
        Semantic.id == data.semantic_id, Semantic.user_id == user_id
    ).first()
    if not semantic:
        raise HTTPException(status_code=404, detail="语义不存在或不属于当前用户")

    article = ArticleDraft(
        user_id=user_id,
        semantic_id=data.semantic_id,
        title=data.title,
        content=data.content,
        platform=data.platform,
        status=data.status
    )
    db.add(article)

    # 更新语义的文章计数
    semantic.article_count = (semantic.article_count or 0) + 1

    # 自动跑实体一致性校验
    profile = db.query(DoctorProfile).filter(DoctorProfile.user_id == user_id).first()
    if profile:
        result = entity_validator.validate_article(article, profile)
        article.entity_consistent = result.passed

    db.commit()
    db.refresh(article)
    return {"code": 0, "message": "success", "data": ArticleOut.model_validate(article).model_dump()}


@router.get("/{article_id}", summary="获取文章详情")
def get_article(
    article_id: int,
    db: Session = Depends(get_db),
    user_id: int = Depends(get_current_user_id)
):
    article = db.query(ArticleDraft).filter(
        ArticleDraft.id == article_id, ArticleDraft.user_id == user_id
    ).first()
    if not article:
        raise HTTPException(status_code=404, detail="文章不存在")
    return {"code": 0, "message": "success", "data": ArticleOut.model_validate(article).model_dump()}


@router.put("/{article_id}", summary="更新文章")
def update_article(
    article_id: int,
    data: ArticleUpdate,
    db: Session = Depends(get_db),
    user_id: int = Depends(get_current_user_id)
):
    article = db.query(ArticleDraft).filter(
        ArticleDraft.id == article_id, ArticleDraft.user_id == user_id
    ).first()
    if not article:
        raise HTTPException(status_code=404, detail="文章不存在")

    update_data = data.model_dump(exclude_unset=True)

    # 如果改了semantic_id，需要更新两个语义的计数
    if "semantic_id" in update_data and update_data["semantic_id"] != article.semantic_id:
        old_semantic = db.query(Semantic).filter(Semantic.id == article.semantic_id).first()
        new_semantic = db.query(Semantic).filter(
            Semantic.id == update_data["semantic_id"], Semantic.user_id == user_id
        ).first()
        if not new_semantic:
            raise HTTPException(status_code=404, detail="目标语义不存在")
        if old_semantic:
            old_semantic.article_count = max(0, (old_semantic.article_count or 0) - 1)
        new_semantic.article_count = (new_semantic.article_count or 0) + 1

    for key, value in update_data.items():
        setattr(article, key, value)

    # 重新跑校验
    profile = db.query(DoctorProfile).filter(DoctorProfile.user_id == user_id).first()
    if profile:
        result = entity_validator.validate_article(article, profile)
        article.entity_consistent = result.passed

    db.commit()
    db.refresh(article)
    return {"code": 0, "message": "success", "data": ArticleOut.model_validate(article).model_dump()}


@router.delete("/{article_id}", summary="删除文章")
def delete_article(
    article_id: int,
    db: Session = Depends(get_db),
    user_id: int = Depends(get_current_user_id)
):
    article = db.query(ArticleDraft).filter(
        ArticleDraft.id == article_id, ArticleDraft.user_id == user_id
    ).first()
    if not article:
        raise HTTPException(status_code=404, detail="文章不存在")

    # 更新语义计数
    semantic = db.query(Semantic).filter(Semantic.id == article.semantic_id).first()
    if semantic:
        semantic.article_count = max(0, (semantic.article_count or 0) - 1)

    db.delete(article)
    db.commit()
    return {"code": 0, "message": "success"}


@router.post("/{article_id}/validate", summary="重新校验文章实体一致性")
def validate_article(
    article_id: int,
    db: Session = Depends(get_db),
    user_id: int = Depends(get_current_user_id)
):
    article = db.query(ArticleDraft).filter(
        ArticleDraft.id == article_id, ArticleDraft.user_id == user_id
    ).first()
    if not article:
        raise HTTPException(status_code=404, detail="文章不存在")

    profile = db.query(DoctorProfile).filter(DoctorProfile.user_id == user_id).first()
    if not profile:
        raise HTTPException(status_code=400, detail="请先完善医生资料")

    result = entity_validator.validate_article(article, profile)
    article.entity_consistent = result.passed
    db.commit()

    return {"code": 0, "message": "success", "data": result.to_dict()}
```

```tsx
// 在useEffect里加载
const loadEntityScore = async () => {
  try {
    const res: any = await entityApi.getProfileScore()
    setEntityScore(res.data)
  } catch (e) {}
}
// 在加载profile的useEffect里调用loadEntityScore()

// 在页面顶部（page-header之后）加评分卡片
{entityScore && (
  <div className="geo-card" style={{ marginBottom: 16, borderLeft: `4px solid ${entityScore.score >= 80 ? '#10B981' : entityScore.score >= 60 ? '#F59E0B' : '#EF4444'}` }}>
    <div style={{ display: 'flex', alignItems: 'center', gap: 24 }}>
      <div style={{ textAlign: 'center' }}>
        <div style={{ fontSize: 36, fontWeight: 700, color: entityScore.score >= 80 ? '#10B981' : entityScore.score >= 60 ? '#F59E0B' : '#EF4444' }}>
          {entityScore.score}
        </div>
        <div style={{ fontSize: 12, color: '#64748B' }}>实体一致性评分</div>
      </div>
      <div style={{ flex: 1 }}>
        <div style={{ fontWeight: 500, marginBottom: 8 }}>
          {entityScore.passed ? (
            <span style={{ color: '#10B981' }}><CheckCircleOutlined /> 实体信息完整，AI识别无歧义</span>
          ) : (
            <span style={{ color: '#F59E0B' }}><WarningOutlined /> 存在 {entityScore.issues.filter((i:any)=>i.level!=='info').length} 个需要优化的问题</span>
          )}
        </div>
        {entityScore.issues.slice(0, 3).map((issue: any, idx: number) => (
          <div key={idx} style={{ fontSize: 12, color: '#64748B', marginBottom: 2 }}>
            <ExclamationCircleOutlined style={{ marginRight: 4, color: issue.level === 'error' ? '#EF4444' : '#F59E0B' }} />
            {issue.message} — {issue.suggestion}
          </div>
        ))}
      </div>
    </div>
  </div>
)}
```

> 注意：Profile.tsx的具体结构需要先Read确认，评分卡片插在page-header之后、表单之前。loadEntityScore在已有的useEffect里调用。

## 11. 坑表（必须避开）

| # | 坑 | 后果 | 正确做法 |
|---|-----|------|----------|
| 1 | ArticleDraft表已存在，不要新建迁移 | 重复建表报错 | 直接用，init_db已import |
| 2 | article API必须加JWT鉴权 | 所有用户能看到所有文章 | 每个端点都加`Depends(get_current_user_id)`，查询过滤user_id |
| 3 | 创建/删除文章时要同步更新semantic.article_count | 语义页文章数不准 | create时+1，delete时-1，update改semantic_id时两边都改 |
| 4 | entity_validator是规则引擎，不调用LLM | 不要在里面加adapter调用 | 纯正则+字符串匹配，保证稳定 |
| 5 | 工具跳转用window.open新窗口 | 不要用路由跳转 | `window.open(url, '_blank')`，带URLSearchParams拼参数 |
| 6 | Content.tsx是完全重写不是修改 | 旧的contentApi调用要全部删掉 | 直接替换整个文件内容，改用articleApi |
| 7 | Profile.tsx加评分前先Read确认结构 | 插错位置导致布局乱 | 评分卡片插在page-header之后、表单之前 |
| 8 | 前端axios超时已设180000ms | 不要改回120000 | 采集接口需要长超时 |
| 9 | schemas里ArticleOut要加from_attributes=True | model_validate报错 | 和其他Out类保持一致 |
| 10 | 文章工具URL带参数时，工具端可能不支持 | 不影响，忽略未知参数即可 | 用URLSearchParams拼，不要手动拼字符串 |
